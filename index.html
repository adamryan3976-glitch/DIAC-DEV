/** * DIAC Staff Hub — backend * * Security model: * - Login verifies the Google ID token against Google's tokeninfo endpoint * (signature, audience, expiry, email_verified) — never decoded locally only. * - On success the server issues a signed session token (email|role|expiry + HMAC). * - EVERY subsequent call (reads and writes) must present a valid session token. * The token is verified server-side on every request, and email/role are taken * from the verified token — never from client-supplied values. * - HOC-only actions are enforced server-side, re-checked against the roster * (with a short cache) on every call, so removing someone from the staff list * revokes them immediately. * - Client-supplied strings that land in sheet cells are validated against * allowlists / known values and defused if they start with a formula trigger * character, to prevent formula injection. * * One-time setup (see setup checklist sent alongside this file): * 1. Script Properties: set SESSION_SECRET to a long random string. * 2. Add a "Note" header in column J of "App Attendance Absences". * 3. Re-deploy as a new version (Deploy > Manage deployments > Edit > New version). */ const SS = SpreadsheetApp.getActiveSpreadsheet(); const SESSION_TTL_MS = 12 * 60 * 60 * 1000; // 12 hours — a camp day plus buffer const STAFF_CACHE_TTL_SEC = 60; const READ_CACHE_TTL_SEC = 45; const GOOGLE_CLIENT_ID = "241712157669-1huukafbcpsifc4v35op6o80qjb10r0r.apps.googleusercontent.com"; const ALLOWED_DOMAINS = ["ddsb.ca", "ddsb-diac.ca"]; const ALLOWED_STATUSES = ["Present", "Absent - Unknown", "Absent - Excused", "Late", "Resolved", "Not Completed"]; // ============================================================================ // Entry points // ============================================================================ function doGet(e) { return handleRequest(e, 'GET'); } function doPost(e) { return handleRequest(e, 'POST'); } function handleRequest(e, method) { const action = e.parameter.action; let data; try { if (action === 'login') { data = checkLogin(e.parameter.token); return respond(data); } // Every other action requires a valid session. const auth = requireSession(e.parameter.session); if (!auth.valid) { return respond({ error: "Unauthorized: " + auth.error }, 401); } switch (action) { case 'students': data = withCache('students', () => getStudentData()); break; case 'schedule': data = withCache('schedule', () => getSheetData("Daily Schedule")); break; case 'absences': data = withCache('absences', () => getSheetData("App Attendance Absences")); break; case 'classes': data = withCache('classes', () => getClassesData()); break; case 'attStudents': data = withCache('attStudents', () => getAttendanceStudents()); break; case 'massageSchedule': data = withCache('massageSchedule', () => getMassageSchedule()); break; case 'updateAbsence': if (!auth.isHoc) { data = { success: false, error: "Forbidden: HOC role required." }; return respond(data, 403); } data = updateAbsenceStatus(e.parameter.id, e.parameter.status, e.parameter.note, auth.email); invalidateCache('absences'); break; case 'bulkReport': { if (method !== 'POST') { data = { success: false, error: "bulkReport requires POST." }; return respond(data, 405); } const rawData = JSON.parse(e.parameter.data); data = bulkReportAbsences(rawData, auth.email); invalidateCache('absences'); break; } default: data = { error: "Invalid action" }; } } catch (err) { data = { error: "Server error: " + err.toString() }; } return respond(data); } function respond(data) { // ContentService responses from a web app can't set arbitrary HTTP status // codes when called with mode:'no-cors' or via simple GET/POST from the // front end, so errors are signaled in the JSON body (data.error / // data.success === false) and the front end must check for them. return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ============================================================================ // Authentication: login // ============================================================================ /** * Validates the Google SSO ID token via Google's tokeninfo endpoint (real * signature verification, not a local decode), checks audience + domain, * looks up the caller on the staff roster, and issues a signed session token. */ function checkLogin(token) { if (!token) { return { success: false, error: "Missing authentication credentials." }; } try { const response = UrlFetchApp.fetch( "https://oauth2.googleapis.com/tokeninfo?id_token=" + encodeURIComponent(token), { muteHttpExceptions: true } ); if (response.getResponseCode() !== 200) { return { success: false, error: "Invalid or expired Google session. Please sign in again." }; } const payload = JSON.parse(response.getContentText()); if (!payload || !payload.email) { return { success: false, error: "Failed to extract SSO identity." }; } if (payload.aud !== GOOGLE_CLIENT_ID) { return { success: false, error: "Token was not issued for this application." }; } if (String(payload.email_verified).toLowerCase() !== 'true') { return { success: false, error: "Email address is not verified with Google." }; } const emailToCheck = payload.email.trim().toLowerCase(); const hostedDomain = (payload.hd || '').trim().toLowerCase(); const isDomainAllowed = ALLOWED_DOMAINS.includes(hostedDomain) || ALLOWED_DOMAINS.some(domain => emailToCheck.endsWith("@" + domain)); if (!isDomainAllowed) { return { success: false, error: "Access Denied: You must sign in using your official school board account." }; } const staff = lookupStaff(emailToCheck); if (!staff) { return { success: false, error: "Access Denied: Email address '" + emailToCheck + "' is not on the authorized App Staff List." }; } const session = createSessionToken(emailToCheck, staff.isHoc); return { success: true, email: emailToCheck, isHoc: staff.isHoc, session: session }; } catch (err) { return { success: false, error: "Google SSO verification failed: " + err.toString() }; } } /** * Looks up a staff email on "App Staff List" (Column A: email, Column F: isHoc). * Cached briefly so a busy login burst doesn't hammer the sheet, but short * enough that removing someone from the roster takes effect within a minute. */ function lookupStaff(emailLower) { const cache = CacheService.getScriptCache(); const cacheKey = 'staff_' + emailLower; const cached = cache.get(cacheKey); if (cached !== null) { return cached === '__none__' ? null : JSON.parse(cached); } const sheet = SS.getSheetByName("App Staff List"); if (!sheet) throw new Error("Database authentication node offline."); const data = sheet.getDataRange().getValues(); let result = null; for (let i = 1; i < data.length; i++) { const rowEmail = String(data[i][0]).toLowerCase().trim(); if (rowEmail === emailLower) { result = { isHoc: String(data[i][5]).toUpperCase() === "TRUE" }; break; } } cache.put(cacheKey, result ? JSON.stringify(result) : '__none__', STAFF_CACHE_TTL_SEC); return result; } // ============================================================================ // Session tokens (HMAC-signed, server-verified on every call) // ============================================================================ function getSessionSecret() { const secret = PropertiesService.getScriptProperties().getProperty('SESSION_SECRET'); if (!secret) { throw new Error("SESSION_SECRET is not configured in Script Properties. See setup checklist."); } return secret; } function createSessionToken(email, isHoc) { const expiry = Date.now() + SESSION_TTL_MS; const payload = email + '|' + (isHoc ? '1' : '0') + '|' + expiry; const sig = signPayload(payload); return Utilities.base64EncodeWebSafe(payload) + '.' + sig; } function signPayload(payload) { const secret = getSessionSecret(); const rawSig = Utilities.computeHmacSha256Signature(payload, secret); return Utilities.base64EncodeWebSafe(rawSig); } /** * Verifies a session token's signature and expiry, then re-checks the * caller's HOC status against the live roster (cached briefly) rather than * trusting the flag embedded in the token — so a demotion takes effect * without waiting for the token to expire. */ function requireSession(token) { if (!token) return { valid: false, error: "No session provided. Please sign in again." }; let payload; try { payload = Utilities.newBlob(Utilities.base64DecodeWebSafe(token.split('.')[0])).getDataAsString(); } catch (e) { return { valid: false, error: "Malformed session." }; } const parts = token.split('.'); if (parts.length !== 2) return { valid: false, error: "Malformed session." }; const sig = parts[1]; const expectedSig = signPayload(payload); if (sig !== expectedSig) { return { valid: false, error: "Invalid session signature." }; } const fields = payload.split('|'); if (fields.length !== 3) return { valid: false, error: "Malformed session." }; const email = fields[0]; const expiry = Number(fields[2]); if (!expiry || Date.now() > expiry) { return { valid: false, error: "Session expired. Please sign in again." }; } const staff = lookupStaff(email); if (!staff) { return { valid: false, error: "You are no longer on the authorized staff list." }; } return { valid: true, email: email, isHoc: staff.isHoc }; } // ============================================================================ // Read-side caching (short TTL; invalidated on writes) // ============================================================================ function withCache(key, fn) { const cache = CacheService.getScriptCache(); const cached = cache.get('data_' + key); if (cached !== null) return JSON.parse(cached); const result = fn(); try { cache.put('data_' + key, JSON.stringify(result), READ_CACHE_TTL_SEC); } catch (e) { // Result too large to cache — fine, just skip caching it. } return result; } function invalidateCache(key) { CacheService.getScriptCache().remove('data_' + key); } // ============================================================================ // Input validation / formula-injection defense // ============================================================================ function sanitizeCell(value) { if (typeof value !== 'string') return value; if (/^[=+\-@]/.test(value)) { return "'" + value; // neutralize as text, matches common spreadsheet defense } return value; } function validateStatus(status) { return ALLOWED_STATUSES.includes(status) ? status : null; } // ============================================================================ // Data access // ============================================================================ function getStudentData() { const sheet = SS.getSheetByName("Student App Data"); const data = sheet.getDataRange().getValues(); return data.slice(1).map(row => ({ firstName: row[0], lastName: row[1], birthDate: formatDate(row[3]), pronouns: row[5], cabin: row[6], guardianFirstName: row[7], guardianLastName: row[8], guardianPhone: row[9], guardianEmail: row[10], preferredEmail: row[11], grade: row[12], school: row[13], photo: row[14], dietary: row[15], allergies: row[16], s1Activity: row[17], s1Venue: row[18], s2Activity: row[19], s2Venue: row[20], s3Activity: row[21], s3Venue: row[22], s4Activity: row[23], s4Venue: row[24], preferredName: row[27] || (row[0] + " " + row[1]), stuNum: String(row[0] + row[1]) })); } function getClassesData() { const sheet = SS.getSheetByName("App Classes"); const data = sheet.getDataRange().getValues(); const result = { S1: [], S2: [], S3: [], S4: [] }; for (let col = 0; col < 4; col++) { for (let row = 1; row < data.length; row++) { if (data[row][col]) result[`S${col + 1}`].push(data[row][col]); } } return result; } /** * Reads the "Massage Appts" tab. Columns are Day / Time / Appt (name) / Email * (a 5th "Notes" column shows up on at least one row in the source sheet and * is intentionally ignored here). The Day column uses merged cells in the * sheet, which Sheets reports as blank for every row after the first in the * merge — so the day is carried forward until a new one appears. Only name * and time are returned; email addresses are left out of the app response * even though the column exists. */ function getMassageSchedule() { const sheet = SS.getSheetByName("Massage Appts"); if (!sheet) return []; const data = sheet.getDataRange().getValues(); const tz = Session.getScriptTimeZone(); let currentDay = ''; const result = []; for (let i = 1; i < data.length; i++) { // skip header row const row = data[i]; const dayCell = formatMassageCell(row[0], tz, false); const timeCell = formatMassageCell(row[1], tz, true); const nameCell = String(row[2] || '').trim(); if (dayCell) currentDay = dayCell; if (!timeCell && !nameCell) continue; // skip blank/separator rows result.push({ day: currentDay, time: timeCell, name: nameCell }); } return result; } function formatMassageCell(value, tz, isTime) { if (value === '' || value === null || value === undefined) return ''; if (Object.prototype.toString.call(value) === '[object Date]') { return Utilities.formatDate(value, tz, isTime ? 'h:mm a' : 'EEEE, MMM d'); } return String(value).trim(); } function getAttendanceStudents() { const sheet = SS.getSheetByName("App Attendance Students"); const data = sheet.getDataRange().getValues(); return data.slice(1).map(row => ({ displayName: row[0], S1: row[1], S2: row[3], S3: row[5], S4: row[7] })); } /** * Applies a batch of attendance updates inside a script lock, so two staff * submitting concurrently (or a double-tap from a no-cors call with no * feedback) can't create duplicate rows. Reads the sheet only after the * lock is acquired, and batches writes with setValues instead of one * setValue call per cell. */ function bulkReportAbsences(updates, reportedByEmail) { const lock = LockService.getScriptLock(); const gotLock = lock.tryLock(10000); if (!gotLock) { return { success: false, error: "System busy, please try again in a moment." }; } try { const sheet = SS.getSheetByName("App Attendance Absences"); const data = sheet.getDataRange().getValues(); const timestamp = new Date(); const todayStr = Utilities.formatDate(timestamp, Session.getScriptTimeZone(), "yyyy-MM-dd"); const rowsToAppend = []; updates.forEach(upd => { const status = validateStatus(upd.status); if (!status) return; // silently skip anything not on the allowlist const studentName = sanitizeCell(String(upd.studentName || '')); const cabin = sanitizeCell(String(upd.cabin || '')); const activity = sanitizeCell(String(upd.activity || '')); const session = String(upd.session || ''); const studentId = sanitizeCell(String(upd.studentId || '000')); let rowIndex = -1; for (let i = 1; i < data.length; i++) { const rowDateStr = Utilities.formatDate(new Date(data[i][7]), Session.getScriptTimeZone(), "yyyy-MM-dd"); if (data[i][2] === studentName && String(data[i][4]) === session && data[i][5] === activity && rowDateStr === todayStr) { rowIndex = i + 1; break; } } if (rowIndex > -1) { sheet.getRange(rowIndex, 7, 1, 3).setValues([[reportedByEmail, timestamp, status]]); } else { const id = Utilities.getUuid(); rowsToAppend.push([id, studentId, studentName, cabin, session, activity, reportedByEmail, timestamp, status, '']); } }); if (rowsToAppend.length > 0) { const startRow = sheet.getLastRow() + 1; sheet.getRange(startRow, 1, rowsToAppend.length, 10).setValues(rowsToAppend); } return { success: true }; } finally { lock.releaseLock(); } } function updateAbsenceStatus(id, status, note, updatedByEmail) { const validStatus = validateStatus(status); if (!validStatus) { return { success: false, error: "Invalid status." }; } const lock = LockService.getScriptLock(); const gotLock = lock.tryLock(10000); if (!gotLock) { return { success: false, error: "System busy, please try again in a moment." }; } try { const sheet = SS.getSheetByName("App Attendance Absences"); const data = sheet.getDataRange().getValues(); const cleanNote = sanitizeCell(String(note || '')); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(id)) { // Columns: 7 timestamp, 9 status, 10 note sheet.getRange(i + 1, 8).setValue(new Date()); sheet.getRange(i + 1, 9).setValue(validStatus); sheet.getRange(i + 1, 10).setValue(cleanNote); return { success: true }; } } return { success: false, error: "Record not found." }; } finally { lock.releaseLock(); } } function getSheetData(name) { const sheet = SS.getSheetByName(name); if (!sheet) return []; const data = sheet.getDataRange().getValues(); const headers = data[0]; return data.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); } function formatDate(date) { if (!date) return ""; try { const d = new Date(date); if (isNaN(d.getTime())) return date; return Utilities.formatDate(d, Session.getScriptTimeZone(), "MMM dd, yyyy"); } catch (e) { return date; } }
